3. Modelo

Crea model/Reserva.java:

package cl.adumun.recintos.model;

public final class Reserva {
    private final String responsable;
    private final String recinto;
    private final String hora;

    public Reserva(String responsable, String recinto, String hora) {
        this.responsable = responsable;
        this.recinto = recinto;
        this.hora = hora;
    }

    public String getResponsable() { return responsable; }
    public String getRecinto() { return recinto; }
    public String getHora() { return hora; }
}

4. Repositorio en memoria

Crea repository/ReservaRepository.java:

package cl.adumun.recintos.repository;

import cl.adumun.recintos.model.Reserva;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;

public final class ReservaRepository {
    private final ObservableList<Reserva> reservas = FXCollections.observableArrayList();

    public ObservableList<Reserva> obtenerTodas() { return reservas; }

    public boolean estaOcupado(String recinto, String hora) {
        return reservas.stream().anyMatch(r ->
                r.getRecinto().equalsIgnoreCase(recinto.trim())
                && r.getHora().equals(hora));
    }

    public void registrar(Reserva reserva) { reservas.add(reserva); }
}

Checkpoint B: ambos archivos compilan. Explica: Reserva representa datos; ReservaRepository administra una colección observable; ninguno debe abrir ventanas. No instancies repositorios distintos por controlador.

2. Archivo controller/ListaController.java

package cl.adumun.recintos.controller;

import cl.adumun.recintos.model.Reserva;
import cl.adumun.recintos.repository.ReservaRepository;
import javafx.beans.property.ReadOnlyStringWrapper;
import javafx.fxml.FXML;
import javafx.scene.control.TableColumn;
import javafx.scene.control.TableView;

public class ListaController {
    @FXML private TableView<Reserva> tablaReservas;
    @FXML private TableColumn<Reserva, String> colResponsable;
    @FXML private TableColumn<Reserva, String> colRecinto;
    @FXML private TableColumn<Reserva, String> colHora;
    private Runnable abrirFormulario;

    public void configurar(ReservaRepository repositorio, Runnable abrirFormulario) {
        this.abrirFormulario = abrirFormulario;
        colResponsable.setCellValueFactory(c ->
                new ReadOnlyStringWrapper(c.getValue().getResponsable()));
        colRecinto.setCellValueFactory(c ->
                new ReadOnlyStringWrapper(c.getValue().getRecinto()));
        colHora.setCellValueFactory(c ->
                new ReadOnlyStringWrapper(c.getValue().getHora()));
        tablaReservas.setItems(repositorio.obtenerTodas());
    }

    @FXML private void abrirFormulario() { abrirFormulario.run(); }
}

2. Crear controller/FormController.java

package cl.adumun.recintos.controller;

import cl.adumun.recintos.model.Reserva;
import cl.adumun.recintos.repository.ReservaRepository;
import javafx.fxml.FXML;
import javafx.scene.control.ComboBox;
import javafx.scene.control.Label;
import javafx.scene.control.TextField;

public class FormController {
    @FXML private TextField campoResponsable;
    @FXML private ComboBox<String> selectorRecinto;
    @FXML private ComboBox<String> selectorHora;
    @FXML private Label mensajeError;
    private ReservaRepository repositorio;
    private Runnable volverAlListado;

    @FXML private void initialize() {
        selectorRecinto.getItems().addAll(
                "Cancha fútbol 1", "Cancha fútbol 2", "Gimnasio");
        for (int h = 8; h < 22; h++) {
            selectorHora.getItems().add(String.format("%02d:00", h));
        }
    }

    public void configurar(ReservaRepository repositorio, Runnable volverAlListado) {
        this.repositorio = repositorio;
        this.volverAlListado = volverAlListado;
    }

    @FXML private void guardar() {
        String responsable = campoResponsable.getText().trim();
        String recinto = selectorRecinto.getValue();
        String hora = selectorHora.getValue();

        if (responsable.isBlank()) {
            mensajeError.setText("Ingresa un responsable");
            return;
        }
        if (recinto == null || hora == null) {
            mensajeError.setText("Selecciona recinto y horario");
            return;
        }
        if (repositorio.estaOcupado(recinto, hora)) {
            mensajeError.setText("El recinto ya está reservado a esa hora");
            return;
        }
        repositorio.registrar(new Reserva(responsable, recinto, hora));
        volverAlListado.run();
    }

    @FXML private void volver() { volverAlListado.run(); }
}

3. Adaptar la clase Application generada

Reemplaza la clase HelloApplication existente por este código, no crees una segunda Application. Ajusta package según IntelliJ y comprueba que los FXML estén en el mismo path de resources que hello-view.fxml:

package cl.adumun.recintos;

import cl.adumun.recintos.controller.FormController;
import cl.adumun.recintos.controller.ListaController;
import cl.adumun.recintos.repository.ReservaRepository;
import javafx.application.Application;
import javafx.fxml.FXMLLoader;
import javafx.scene.Parent;
import javafx.scene.Scene;
import javafx.stage.Stage;
import java.io.IOException;

public class HelloApplication extends Application {
    private final ReservaRepository repositorio = new ReservaRepository();
    private Stage stage;

    @Override public void start(Stage stage) throws IOException {
        this.stage = stage;
        stage.setTitle("Arriendo de recintos deportivos");
        mostrarListado();
        stage.show();
    }

    private void mostrarListado() throws IOException {
        FXMLLoader loader = new FXMLLoader(
                HelloApplication.class.getResource("lista-view.fxml"));
        Parent root = loader.load();
        ListaController controller = loader.getController();
        controller.configurar(repositorio, this::irAFormulario);
        stage.setScene(new Scene(root));
    }

    private void mostrarFormulario() throws IOException {
        FXMLLoader loader = new FXMLLoader(
                HelloApplication.class.getResource("form-view.fxml"));
        Parent root = loader.load();
        FormController controller = loader.getController();
        controller.configurar(repositorio, this::irAListado);
        stage.setScene(new Scene(root));
    }

    private void irAFormulario() {
        try { mostrarFormulario(); }
        catch (IOException e) { throw new IllegalStateException("Carga formulario", e); }
    }

    private void irAListado() {
        try { mostrarListado(); }
        catch (IOException e) { throw new IllegalStateException("Carga listado", e); }
    }

    public static void main(String[] args) { launch(args); }
}

Dos reservas compatibles; una repetida rechazada; cancelar no crea registro.

Prueba guiada de principio a fin

  1. Inicia ReservaRecintosFX desde IntelliJ.
  2. Comprueba que ves una tabla vacía con columnas Responsable, Recinto y Hora.
  3. Haz clic en Nueva reserva.
  4. Escribe Ana, selecciona Cancha fútbol 1 y 09:00; pulsa Guardar.
  5. Verifica que regresa al listado y aparece exactamente una fila.
  6. Entra otra vez al formulario y escribe Pedro, Cancha fútbol 1, 09:00.
  7. Al pulsar Guardar, debe mostrar que el horario está ocupado, sin abandonar el formulario.
  8. Cambia el recinto a Cancha fútbol 2 y pulsa Guardar. Ahora deben aparecer dos filas.
  9. Abre el formulario, deja responsable vacío y confirma que no registra otra reserva.
  10. Cierra y vuelve a ejecutar. La tabla volverá a estar vacía: este laboratorio trabaja solo en memoria. La persistencia JSON se aborda en otro proyecto autónomo.
